// =================================================== // 設定エリア — ここだけ変更してください // =================================================== const SHEET_NAME = "予約データ"; // 予約データシート名 const SCHEDULE_SHEET = "日付別スケジュール"; // 管理用スケジュールシート名 const ADMIN_EMAIL = "tsuchi.awaji@gmail.com,shido@kinkikabezai.com"; // 管理者の通知先メールアドレス（複数の場合はカンマ区切り） const MAX_GROUPS_WEEK = 2; // 平日：1枠あたりの最大組数 const MAX_GROUPS_HOLI = 1; // 土日祝：1枠あたりの最大組数 const MAX_PEOPLE = 4; // 1組あたりの最大人数 const EVENT_START = "2026-07-04"; // 会期開始日 const EVENT_END = "2026-08-16"; // 会期終了日 const SLOTS_WEEKDAY = ["13:00","14:00","15:00","16:00"]; // 平日の時間帯 const SLOTS_HOLIDAY = ["10:00","11:00","13:00","14:00","15:00","16:00","17:00"]; // 土日祝の時間帯 const SETTING_SHEET = "設定"; // 特定日組数設定シート名 // 祝日リスト（"YYYY-MM-DD" 形式で追加してください） const HOLIDAYS = [ "2026-07-20","2026-08-11" ]; // =================================================== // Web API エントリポイント（GitHub Pages用・JSONP対応） // =================================================== function doGet(e) { const callback = e.parameter.callback; let result; if (e.parameter.action === "book") { // GETパラメータから予約情報を取得 try { const params = { date: e.parameter.date, time: e.parameter.time, name: e.parameter.name, email: e.parameter.email, tel: e.parameter.tel || "", count: e.parameter.count || 1, note: e.parameter.note || "" }; result = createBooking(params); } catch(err) { result = { error: err.message }; } } else { // 空き状況取得 result = getAvailability(e.parameter.weekStart); } // JSONP形式で返す（CORSを回避） const json = JSON.stringify(result); const output = callback ? ContentService.createTextOutput(callback + "(" + json + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT) : ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); return output; } // =================================================== // クライアントから呼び出す関数 // =================================================== /** 指定週のうち会期内の空き状況を返す */ function getAvailability(weekStart) { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); // 特別設定シートから特定日の組数を読み込む const specialGroups = getSpecialGroups(); const result = {}; const start = new Date(weekStart); for (let i = 0; i < 7; i++) { const d = new Date(start); d.setDate(start.getDate() + i); const ds = fmtDate(d); // 会期外の日は含めない if (ds < EVENT_START || ds > EVENT_END) continue; const slots = getSlotsForDate(ds); const defaultGroups = isHoliday(ds) ? MAX_GROUPS_HOLI : MAX_GROUPS_WEEK; const daySpec = specialGroups[ds] || {}; result[ds] = {}; slots.forEach(t => { // 時間帯指定 → 日付全体指定 → デフォルトの優先順 const maxGroups = daySpec[t] !== undefined ? daySpec[t] : daySpec["__all__"] !== undefined ? daySpec["__all__"] : defaultGroups; result[ds][t] = { groups: 0, maxGroups: maxGroups, people: 0 }; }); } for (let r = 1; r < data.length; r++) { const row = data[r]; const ds = fmtDateFromCell(row[0]); const rawTime = row[1]; // B列がDate型の場合は HH:MM に変換、文字列ならそのまま let t; if (rawTime instanceof Date) { const h = rawTime.getHours(); const m = rawTime.getMinutes(); t = String(h).padStart(2, '0') + ':' + String(m).padStart(2, '0'); } else { t = String(rawTime).trim(); } const cnt = parseInt(row[6]) || 1; const status = row[8]; if (status === "キャンセル") continue; if (result[ds] && result[ds][t] !== undefined) { result[ds][t].groups += 1; result[ds][t].people += cnt; } } return result; } /** 予約を登録し、スケジュールシートを更新する */ function createBooking(params) { const { date, time, name, email, tel, count, note } = params; const cnt = parseInt(count) || 1; // 会期内チェック if (date < EVENT_START || date > EVENT_END) { return { ok: false, error: "会期外の日付です" }; } const validSlots = getSlotsForDate(date); if (!validSlots.includes(time)) { return { ok: false, error: "この日時は受け付けていません" }; } if (cnt > MAX_PEOPLE) return { ok: false, error: "1組あたりの最大人数は" + MAX_PEOPLE + "名までです" }; const avail = getAvailability(getMondayOf(date)); const slot = avail[date] && avail[date][time]; if (!slot) return { ok: false, error: "該当するコマが見つかりません" }; if (slot.groups >= slot.maxGroups) return { ok: false, error: "満席になりました。別の日時をお選びください。" }; // 念のため最新の組数上限も確認 const specialGroups2 = getSpecialGroups(); const daySpec2 = specialGroups2[date] || {}; const maxG = daySpec2[time] !== undefined ? daySpec2[time] : daySpec2["__all__"] !== undefined ? daySpec2["__all__"] : (isHoliday(date) ? MAX_GROUPS_HOLI : MAX_GROUPS_WEEK); if (slot.groups >= maxG) return { ok: false, error: "満席になりました。別の日時をお選びください。" }; const id = "BK-" + Date.now().toString(36).toUpperCase() + "-" + Math.floor(Math.random() * 1000); const sheet = getSheet(); sheet.appendRow([ date, time, name, email, tel || "", new Date(), cnt, note || "", "確定", id ]); // 日付別スケジュールシートを更新 updateScheduleSheet(); // メール送信（メールアドレスがある場合のみ） if (email && email.includes("@")) { sendConfirmationEmail(email, name, date, time, cnt, id, tel || "", note || ""); } // 管理者通知 MailApp.sendEmail({ to: ADMIN_EMAIL, subject: `【新規予約】TSUCHI展覧会 ${fmtJa(date)} ${time} ／ ${name} 様`, body: `新しい予約が入りました。\n\n日時: ${fmtJa(date)} ${time}\n氏名: ${name}\nメール: ${email || "なし"}\n電話番号: ${tel || "なし"}\n参加人数: ${cnt} 名\n備考: ${note || "なし"}\n予約ID: ${id}` }); // Googleカレンダーに予定を追加してイベントIDを保存 var calEventId = addToCalendar(date, time, name, cnt, tel || "", note || "", id); if (calEventId) { var lastRow = sheet.getLastRow(); sheet.getRange(lastRow, 11).setValue(calEventId); // K列にカレンダーID保存 } return { ok: true, id }; } // =================================================== // 日付別スケジュールシートの生成・更新 // =================================================== function updateScheduleSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let schSheet = ss.getSheetByName(SCHEDULE_SHEET); if (!schSheet) { schSheet = ss.insertSheet(SCHEDULE_SHEET); } schSheet.clearContents(); schSheet.clearFormats(); // 予約データを全取得 const dataSheet = getSheet(); const rows = dataSheet.getDataRange().getValues(); // 日付→時間帯→予約リスト の構造に整理 const map = {}; for (let r = 1; r < rows.length; r++) { const row = rows[r]; const ds = fmtDateFromCell(row[0]); const rawTime = row[1]; // B列がDate型の場合は HH:MM に変換 let t; if (rawTime instanceof Date) { const h = rawTime.getHours(); const m = rawTime.getMinutes(); t = String(h).padStart(2, '0') + ':' + String(m).padStart(2, '0'); } else { t = String(rawTime).trim(); } const name = row[2]; const cnt = parseInt(row[6]) || 1; const note2 = row[7] || ""; const status = row[8]; if (status === "キャンセル") continue; if (!map[ds]) map[ds] = {}; if (!map[ds][t]) map[ds][t] = []; const note = row[7] || ""; map[ds][t].push({ name, cnt, note: note2 }); } // 会期全日付を生成 const dates = []; const cur = new Date(EVENT_START); const end = new Date(EVENT_END); while (cur <= end) { dates.push(fmtDate(new Date(cur))); cur.setDate(cur.getDate() + 1); } const DAY_NAMES = ["日","月","火","水","木","金","土"]; let outRow = 1; dates.forEach(ds => { const d = new Date(ds); const dow = DAY_NAMES[d.getDay()]; const isHoli = d.getDay() === 0 || d.getDay() === 6 || HOLIDAYS.includes(ds); const slots = getSlotsForDate(ds); const [y,m,day] = ds.split("-"); // 日付ヘッダー行 schSheet.getRange(outRow, 1).setValue(`${parseInt(m)}月${parseInt(day)}日（${dow}）`); const hdrRange = schSheet.getRange(outRow, 1, 1, 5); hdrRange.setBackground(isHoli ? "#f0e6f6" : "#e8f0fe"); hdrRange.setFontWeight("bold"); hdrRange.setFontSize(11); outRow++; // 列ヘッダー schSheet.getRange(outRow, 1, 1, 6).setValues([["時間帯","予約者","人数","合計人数","残席","備考"]]); schSheet.getRange(outRow, 1, 1, 6).setBackground("#f8f8f8").setFontWeight("bold").setFontSize(10); outRow++; slots.forEach(t => { const bookings = (map[ds] && map[ds][t]) || []; const totalCnt = bookings.reduce((s, b) => s + b.cnt, 0); const remaining = MAX_PEOPLE * (isHoliday(ds) ? MAX_GROUPS_HOLI : MAX_GROUPS_WEEK) - totalCnt; if (bookings.length === 0) { const maxG = isHoliday(ds) ? MAX_GROUPS_HOLI : MAX_GROUPS_WEEK; schSheet.getRange(outRow, 1, 1, 6).setValues([[t, "（なし）", "", 0, maxG + "組", ""]]); schSheet.getRange(outRow, 5).setFontColor("#888888"); outRow++; } else { // 最初の予約者 const maxG2 = isHoliday(ds) ? MAX_GROUPS_HOLI : MAX_GROUPS_WEEK; const remGroups = maxG2 - bookings.length; const remLabel = remGroups <= 0 ? "満席" : remGroups + "組"; schSheet.getRange(outRow, 1, 1, 6).setValues([[t, bookings[0].name, bookings[0].cnt + "名", bookings.length + "組/" + maxG2 + "組", remLabel, bookings[0].note || ""]]); if (remGroups <= 0) { schSheet.getRange(outRow, 5).setFontColor("#cc0000").setFontWeight("bold"); } outRow++; // 2人目以降 for (let i = 1; i < bookings.length; i++) { schSheet.getRange(outRow, 1, 1, 4).setValues([["", bookings[i].name, bookings[i].cnt + "名", bookings[i].note || ""]]); outRow++; } } }); // 区切り空行 outRow++; }); // 列幅を整える schSheet.setColumnWidth(1, 100); schSheet.setColumnWidth(2, 160); schSheet.setColumnWidth(3, 60); schSheet.setColumnWidth(4, 80); schSheet.setColumnWidth(5, 60); schSheet.setColumnWidth(6, 200); } // =================================================== // 内部ヘルパー // =================================================== function isHoliday(dateStr) { const d = new Date(dateStr); const dow = d.getDay(); return dow === 0 || dow === 6 || HOLIDAYS.includes(dateStr); } function getSlotsForDate(dateStr) { return isHoliday(dateStr) ? SLOTS_HOLIDAY : SLOTS_WEEKDAY; } /** 「設定」シートから特定日・時間帯の組数設定を読み込む * A列：日付、B列：時間帯（空欄=その日の全枠）、C列：組数 * 戻り値: { "2026-07-04": { "__all__": 4 }, "2026-07-05": { "14:00": 3 } } */ function getSpecialGroups() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SETTING_SHEET); if (!sheet) return {}; const rows = sheet.getDataRange().getValues(); const map = {}; for (let r = 1; r < rows.length; r++) { const ds = fmtDateFromCell(rows[r][0]); const rawTime = rows[r][1]; const cnt = parseInt(rows[r][2]); if (!ds || isNaN(cnt) || cnt < 0) continue; if (!map[ds]) map[ds] = {}; // B列がDate型の場合は HH:MM に変換 let key; if (rawTime instanceof Date) { const h = rawTime.getHours(); const m = rawTime.getMinutes(); key = String(h).padStart(2, '0') + ':' + String(m).padStart(2, '0'); } else { const t = String(rawTime).trim(); key = (t && t !== "" && t !== "0") ? t : "__all__"; } map[ds][key] = cnt; } return map; } /** 「設定」シートを初期化（なければ作成） */ function initSettingSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SETTING_SHEET); if (!sheet) { sheet = ss.insertSheet(SETTING_SHEET); } sheet.clearContents(); sheet.clearFormats(); // ヘッダー sheet.appendRow(["日付（YYYY-MM-DD）", "時間帯（空欄=全枠）", "組数"]); sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, 3).setBackground("#e8f0fe").setFontWeight("bold"); sheet.setColumnWidth(1, 180); sheet.setColumnWidth(2, 180); sheet.setColumnWidth(3, 80); // 使い方の説明（2行目以降はデータ行として使用） sheet.getRange(1, 1).setNote("例：2026-07-04"); sheet.getRange(1, 2).setNote("特定の時間帯だけ変えたい場合に記入（例：14:00）。空欄にするとその日の全枠に適用されます。"); sheet.getRange(1, 3).setNote("その日・その時間帯の最大組数を入力してください。"); return sheet; } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(["日付","時間帯","氏名","メール","電話番号","申込日時","参加人数","備考","ステータス","予約ID","カレンダーID"]); sheet.setFrozenRows(1); } return sheet; } function fmtDate(d) { return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`; } function fmtDateFromCell(val) { if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); } return String(val).slice(0, 10); } function fmtJa(ds) { const [y,m,d] = ds.split("-"); return `${parseInt(m)}月${parseInt(d)}日`; } function pad(n) { return String(n).padStart(2, "0"); } function getMondayOf(dateStr) { const d = new Date(dateStr); const day = d.getDay(); d.setDate(d.getDate() - (day === 0 ? 6 : day - 1)); return fmtDate(d); } function sendConfirmationEmail(email, name, date, time, count, id, tel, note) { const dateJa = fmtJa(date) + " " + time; const subject = `【予約確定】TSUCHI 展覧会 ${dateJa}`; const body = `${name} 様 この度はTSUCHI LAND/TRACE -大地のプロダクト展- へのご予約ありがとうございます。 以下の内容で予約が確定しました。 ━━━━━━━━━━━━━━━━ 　日時: ${dateJa} 　参加人数: ${count} 名 　電話番号: ${tel || "なし"} 　備考: ${note || "なし"} 　予約ID: ${id} ━━━━━━━━━━━━━━━━ ご予約のキャンセル・変更につきましては、下記までお電話にてご連絡ください。 TEL：0799-85-1147 皆様のご来場を心よりお待ちしております。 `; MailApp.sendEmail({ to: email, subject, body }); } // =================================================== // Googleカレンダー連携 // =================================================== function addToCalendar(date, time, name, cnt, tel, note, id) { try { const cal = CalendarApp.getDefaultCalendar(); const [y,m,d] = date.split('-').map(Number); const [h, min] = time.split(':').map(Number); const start = new Date(y, m-1, d, h, min, 0); const end = new Date(y, m-1, d, h+1, min, 0); const title = 'TSUCHI予約 — ' + name + ' 様（' + cnt + '名）'; const desc = [ '予約ID: ' + id, '氏名: ' + name, '人数: ' + cnt + '名', '電話: ' + (tel || 'なし'), '備考: ' + (note || 'なし') ].join('\n'); const event = cal.createEvent(title, start, end, { description: desc }); return event.getId(); // イベントIDを返す } catch(e) { Logger.log('カレンダー追加エラー: ' + e.message); return null; } } /** スプレッドシートのステータスが「キャンセル」に変わったとき * 対応するGoogleカレンダーの予定を削除するトリガー関数 */ function onSheetEdit(e) { const sheet = e.source.getActiveSheet(); if (sheet.getName() !== SHEET_NAME) return; const row = e.range.getRow(); const col = e.range.getColumn(); // I列（ステータス=9列目）が「キャンセル」に変更されたとき if (col === 9 && e.value === "キャンセル") { const calEventId = sheet.getRange(row, 11).getValue(); // K列のカレンダーID if (!calEventId) return; try { const cal = CalendarApp.getDefaultCalendar(); const event = cal.getEventById(calEventId); if (event) { event.deleteEvent(); Logger.log('カレンダー予定を削除しました: ' + calEventId); } } catch(err) { Logger.log('カレンダー削除エラー: ' + err.message); } } }
